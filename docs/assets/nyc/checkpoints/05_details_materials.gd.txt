extends RefCounted
## Original low-poly street kit. All vehicles face +X. Wheel pivots rotate on Z.
static var meshes: Dictionary = {}
static var materials: Dictionary = {}

static func material(color: Color, emission: float = 0.0) -> StandardMaterial3D:
	var key := str(color) + str(emission)
	if not materials.has(key):
		var value := StandardMaterial3D.new()
		value.albedo_color = color
		value.roughness = 0.9
		value.metallic_specular = 0.0
		value.shading_mode = BaseMaterial3D.SHADING_MODE_PER_VERTEX
		if emission > 0:
			value.emission_enabled = true
			value.emission = color
			value.emission_energy_multiplier = emission
		materials[key] = value
	return materials[key]

static func box(parent: Node3D, title: String, size: Vector3, pos: Vector3, color: Color, glow: float = 0.0) -> MeshInstance3D:
	var key := str(size)
	if not meshes.has(key):
		var mesh := BoxMesh.new()
		mesh.size = size
		meshes[key] = mesh
	var piece := MeshInstance3D.new()
	piece.name = title
	piece.mesh = meshes[key]
	piece.material_override = material(color, glow)
	piece.position = pos
	parent.add_child(piece)
	return piece

static func cylinder(parent: Node3D, title: String, radius: float, height: float, pos: Vector3, color: Color, top: float = -1.0) -> MeshInstance3D:
	var key := "cylinder" + str([radius, height, top])
	if not meshes.has(key):
		var mesh := CylinderMesh.new()
		mesh.bottom_radius = radius
		mesh.top_radius = radius if top < 0 else top
		mesh.height = height
		mesh.radial_segments = 8
		meshes[key] = mesh
	var piece := MeshInstance3D.new()
	piece.name = title
	piece.mesh = meshes[key]
	piece.material_override = material(color)
	piece.position = pos
	parent.add_child(piece)
	return piece

static func vehicle_length(kind: String) -> float:
	return 4.4 if kind == "bus" else (3.5 if kind == "truck" else 2.6)

static func vehicle(kind: String, variant: int = 0) -> Node3D:
	var root := Node3D.new()
	root.name = "NYC_" + kind
	var length := vehicle_length(kind)
	var paint := Color("efbc32") if kind == "taxi" else Color("375f84")
	if kind == "bus":
		paint = Color("dce2d9")
	elif kind == "truck":
		paint = Color("be7860")
	elif kind == "sedan":
		paint = [Color("375f84"), Color("bc5a58"), Color("7c947d"), Color("c8c1a4")][posmod(variant, 4)]
	box(root, "Chassis", Vector3(length - 0.1, 0.22, 1.0), Vector3(0, 0.27, 0), Color("252a35"))
	box(root, "Body", Vector3(length, 0.48, 1.08), Vector3(0, 0.54, 0), paint)
	if kind == "bus":
		box(root, "PassengerBody", Vector3(length - 0.16, 0.92, 1.05), Vector3(0, 1.13, 0), paint)
	elif kind == "truck":
		box(root, "CargoBox", Vector3(2.25, 1.08, 1.08), Vector3(-0.53, 1.03, 0), Color("d4cbb4"))
		box(root, "Cab", Vector3(1.03, 0.75, 1.04), Vector3(1.04, 0.90, 0), paint)
	else:
		box(root, "Cabin", Vector3(1.32, 0.50, 0.94), Vector3(-0.1, 0.97, 0), paint.lightened(0.08))
	vehicle_components(root, kind, length)
	vehicle_detail(root, kind, length)
	bake_body(root)
	return root

static func bake_body(root: Node3D) -> void:
	# Merge stationary body assemblies by material. Wheel pivots stay separate.
	var groups: Dictionary = {}
	for piece in root.find_children("*", "MeshInstance3D", true, false):
		if str(piece.get_parent().name).begins_with("Wheel_"):
			continue
		var key: int = piece.material_override.get_instance_id()
		if not groups.has(key):
			groups[key] = {"material": piece.material_override, "pieces": []}
		groups[key].pieces.append(piece)
	for group in groups.values():
		var tool := SurfaceTool.new()
		tool.begin(Mesh.PRIMITIVE_TRIANGLES)
		for piece in group.pieces:
			tool.append_from(piece.mesh, 0, piece.transform)
			root.remove_child(piece)
			piece.free()
		var merged := MeshInstance3D.new()
		merged.name = "BatchedBody"
		merged.mesh = tool.commit()
		merged.material_override = group.material
		root.add_child(merged)

static func vehicle_detail(root: Node3D, kind: String, length: float) -> void:
	var dark := Color("233542")
	for side in [-1, 1]:
		# Mirrors and handles remain bold enough for the low-resolution game camera.
		var cab_x := 1.25 if kind == "truck" else (1.75 if kind == "bus" else 0.42)
		box(root, "MirrorArm", Vector3(0.12, 0.05, 0.15), Vector3(cab_x, 0.95, side * 0.60), dark)
		box(root, "Mirror", Vector3(0.13, 0.17, 0.06), Vector3(cab_x, 0.98, side * 0.69), Color("9cb5bb"))
		if kind not in ["bus", "truck"]:
			for x in [-0.47, 0.15]:
				box(root, "DoorHandle", Vector3(0.12, 0.045, 0.025), Vector3(x, 0.72, side * 0.55), Color("d9d3bc"))
		if kind == "taxi":
			box(root, "CheckerBase", Vector3(1.55, 0.10, 0.025), Vector3(-0.1, 0.61, side * 0.55), Color("eee0a8"))
			for check in range(10):
				box(root, "Checker", Vector3(0.075, 0.055, 0.03), Vector3(-0.8 + check * 0.15, 0.625 if check % 2 == 0 else 0.595, side * 0.57), dark)
		elif kind == "truck":
			box(root, "CargoPanelInset", Vector3(1.95, 0.72, 0.025), Vector3(-0.53, 1.05, side * 0.555), Color("b8ae99"))
			for x in [-1.48, 0.42]:
				box(root, "CargoCorner", Vector3(0.065, 0.94, 0.03), Vector3(x, 1.03, side * 0.58), Color("e4d9ba"))
		elif kind == "bus":
			box(root, "LowerStripe", Vector3(length - 0.18, 0.065, 0.025), Vector3(0, 0.54, side * 0.56), Color("dbad4c"))
	box(root, "Plate", Vector3(0.03, 0.10, 0.25), Vector3(-length * 0.5 - 0.08, 0.36, 0), Color("d8b05b"))
	if kind == "truck":
		for y in range(6):
			box(root, "RollupSlat", Vector3(0.025, 0.025, 0.80), Vector3(-1.72, 0.68 + y * 0.13, 0), Color("7f9294"))
	elif kind == "bus":
		box(root, "RouteDisplay", Vector3(0.03, 0.13, 0.81), Vector3(2.18, 1.49, 0), dark)
		for index in range(7):
			box(root, "RoutePixel", Vector3(0.035, 0.05, 0.055), Vector3(2.20, 1.49, -0.29 + index * 0.095), Color("f0c66c"), 0.3)
	elif kind == "taxi":
		for side in [-1, 1]:
			# Four dark glyph bars suggest the roof lettering at the intended scale.
			for index in range(4):
				box(root, "SignGlyph", Vector3(0.05, 0.09, 0.025), Vector3(-0.28 + index * 0.12, 1.36, side * 0.125), dark)

static func vehicle_components(root: Node3D, kind: String, length: float) -> void:
	var dark := Color("233542")
	var glass := Color("8ab2b9")
	var axle := length * (0.34 if kind in ["bus", "truck"] else 0.32)
	for end in [-1, 1]:
		for side in [-1, 1]:
			var pivot := Node3D.new()
			pivot.name = "Wheel_%s_%s" % ["Front" if end == 1 else "Rear", "Left" if side == -1 else "Right"]
			pivot.position = Vector3(end * axle, 0.29, side * 0.53)
			root.add_child(pivot)
			cylinder(pivot, "Tire", 0.29, 0.16, Vector3.ZERO, Color("202630")).rotation.x = PI / 2
			cylinder(pivot, "Hub", 0.135, 0.175, Vector3.ZERO, Color("9ca8ad")).rotation.x = PI / 2
	box(root, "FrontBumper", Vector3(0.12, 0.13, 1.10), Vector3(length * 0.5 + 0.01, 0.34, 0), Color("c7c6b5"))
	box(root, "RearBumper", Vector3(0.12, 0.13, 1.10), Vector3(-length * 0.5 - 0.01, 0.34, 0), Color("959d9c"))
	box(root, "Grille", Vector3(0.025, 0.16, 0.45), Vector3(length * 0.5 + 0.015, 0.52, 0), dark)
	for side in [-1, 1]:
		box(root, "Headlight", Vector3(0.04, 0.16, 0.19), Vector3(length * 0.5 + 0.025, 0.58, side * 0.38), Color("fff0b7"), 0.65)
		box(root, "TailLight", Vector3(0.035, 0.16, 0.19), Vector3(-length * 0.5 - 0.025, 0.58, side * 0.38), Color("df594b"), 0.3)
	if kind == "bus":
		box(root, "WindshieldFrame", Vector3(0.035, 0.61, 0.94), Vector3(2.14, 1.22, 0), dark)
		box(root, "Windshield", Vector3(0.045, 0.43, 0.82), Vector3(2.16, 1.19, 0), glass)
		for side in [-1, 1]:
			for index in range(7):
				box(root, "PassengerWindow_%d_%d" % [side, index], Vector3(0.40, 0.42, 0.025), Vector3(-1.70 + index * 0.53, 1.25, side * 0.535), dark if index % 3 == 0 else glass)
			box(root, "TransitStripe", Vector3(4.25, 0.20, 0.03), Vector3(0, 0.79, side * 0.55), Color("326a9c"))
		for x in [-0.9, 1.5]:
			box(root, "DoorFrame", Vector3(0.48, 0.92, 0.04), Vector3(x, 0.91, 0.565), dark)
			for section in [-1, 1]:
				box(root, "DoorGlass", Vector3(0.18, 0.58, 0.045), Vector3(x + section * 0.11, 1.02, 0.59), glass)
		box(root, "RoofVent", Vector3(1.10, 0.14, 0.76), Vector3(-0.6, 1.65, 0), Color("a6b2b1"))
	elif kind == "truck":
		box(root, "WindshieldFrame", Vector3(0.03, 0.47, 0.91), Vector3(1.57, 1.01, 0), dark)
		box(root, "Windshield", Vector3(0.04, 0.35, 0.79), Vector3(1.59, 1.02, 0), glass)
		for side in [-1, 1]:
			box(root, "CabWindow", Vector3(0.60, 0.32, 0.035), Vector3(1.04, 1.06, side * 0.535), glass)
			box(root, "DoorSeam", Vector3(0.025, 0.62, 0.025), Vector3(0.65, 0.88, side * 0.548), dark)
		box(root, "RollupFrame", Vector3(0.035, 0.91, 0.93), Vector3(-1.67, 1.02, 0), dark)
		box(root, "RollupDoor", Vector3(0.045, 0.80, 0.82), Vector3(-1.69, 1.02, 0), Color("a7b2b1"))
	else:
		box(root, "Windshield", Vector3(0.035, 0.32, 0.81), Vector3(0.575, 0.98, 0), glass)
		box(root, "RearWindow", Vector3(0.035, 0.29, 0.79), Vector3(-0.775, 0.99, 0), dark)
		for side in [-1, 1]:
			for x in [-0.43, 0.20]:
				box(root, "SideWindow", Vector3(0.53, 0.30, 0.025), Vector3(x, 1.0, side * 0.483), glass)
			box(root, "DoorPillar", Vector3(0.065, 0.40, 0.045), Vector3(-0.10, 0.98, side * 0.49), dark)
	if kind == "taxi":
		box(root, "RoofSignBase", Vector3(0.58, 0.05, 0.30), Vector3(-0.1, 1.25, 0), dark)

static func district(root: Node3D, width: int, height: int, cell: float) -> void:
	var half_width := width * cell * 0.5
	var back := -height * cell * 0.5 - 3.0
	for i in range(-4, 5):
		var building_height := 4.0 + posmod(i * 7 + 17, 6) * 1.1
		building(root, "Building_%d" % i, Vector3(i * 4.5, 0, back), 4.0, building_height, i + 7)
	for side in [-1, 1]:
		for i in range(3):
			building(root, "SideBuilding", Vector3(side * (half_width + 2.7), 0, back + 5.0 + i * 4.0), 3.0, 4.0 + i, i + 2)
	# A stepped spire and two bold billboards form the distant avenue landmark.
	var landmark := Vector3(half_width * 0.35, 0, back - 2.2)
	for tier in range(4):
		box(root, "MidtownSpireTier", Vector3(3.6 - tier * 0.7, 2.2, 3.0 - tier * 0.5), landmark + Vector3(0, 5.5 + tier * 2.0, 0), Color("6c7889"))
	cylinder(root, "MidtownAntenna", 0.07, 2.0, landmark + Vector3(0, 13.8, 0), Color("c1c2b5"), 0.01)
	for index in range(2):
		var sign_pos := Vector3(-half_width * 0.45 + index * 8.0, 3.3 + index * 1.7, back + 2.2)
		box(root, "BillboardFrame", Vector3(4.2, 1.9, 0.16), sign_pos, Color("242d3c"))
		box(root, "BillboardColor", Vector3(3.85, 1.58, 0.05), sign_pos + Vector3(0, 0, 0.1), Color("368a92") if index == 0 else Color("ac526d"), 0.2)
		label(root, "HOP NYC" if index == 0 else "NIGHT SHIFT", sign_pos + Vector3(0, 0, 0.15), 0.008, Color("ffe7a0"))

static func building(parent: Node3D, title: String, pos: Vector3, width: float, height: float, variant: int) -> void:
	var root := Node3D.new()
	root.name = title
	root.position = pos
	parent.add_child(root)
	var masonry: Color = [Color("80695f"), Color("5c6578"), Color("976f66"), Color("747c72")][posmod(variant, 4)]
	box(root, "Masonry", Vector3(width, height, 3.5), Vector3(0, height * 0.5, 0), masonry)
	box(root, "Cornice", Vector3(width + 0.22, 0.18, 3.7), Vector3(0, height, 0), masonry.lightened(0.18))
	box(root, "RoofSetback", Vector3(width * 0.70, 0.55, 2.6), Vector3(0, height + 0.27, -0.2), masonry.darkened(0.18))
	for floor_index in range(1, int(height / 0.85)):
		for bay in range(3):
			var window_color := Color("d5bf7c") if posmod(floor_index * 3 + bay + variant, 4) == 0 else Color("324658")
			box(root, "Window_%d_%d" % [floor_index, bay], Vector3(width * 0.15, 0.47, 0.035), Vector3((bay - 1) * width * 0.28, floor_index * 0.85, 1.77), window_color, 0.10 if window_color.r > 0.5 else 0.0)
			box(root, "WindowLintel", Vector3(width * 0.18, 0.06, 0.09), Vector3((bay - 1) * width * 0.28, floor_index * 0.85 + 0.28, 1.78), masonry.lightened(0.22))
	if posmod(variant, 2) == 0:
		var tower := Node3D.new()
		tower.name = "RooftopWaterTower"
		tower.position = Vector3(-0.45, height + 0.55, 0)
		root.add_child(tower)
		for x in [-0.40, 0.40]:
			for z in [-0.40, 0.40]:
				box(tower, "Support", Vector3(0.075, 0.85, 0.075), Vector3(x, 0.42, z), Color("38444c"))
		cylinder(tower, "WoodTank", 0.66, 1.00, Vector3(0, 1.25, 0), Color("92765b"))
		cylinder(tower, "ConicalRoof", 0.77, 0.38, Vector3(0, 1.94, 0), Color("48515c"), 0.0)
		for y in [0.85, 1.24, 1.63]:
			cylinder(tower, "TankBand", 0.68, 0.045, Vector3(0, y, 0), Color("454a4a"))
	else:
		for floor_index in range(1, int(height / 1.4)):
			var y := floor_index * 1.4
			box(root, "FireEscapeLanding", Vector3(width * 0.72, 0.06, 0.50), Vector3(0, y, 2.0), Color("303e49"))
			box(root, "FireEscapeRail", Vector3(width * 0.72, 0.04, 0.04), Vector3(0, y + 0.43, 2.24), Color("4d5b60"))
			for side in [-1, 1]:
				box(root, "RailPost", Vector3(0.04, 0.43, 0.04), Vector3(side * width * 0.32, y + 0.21, 2.24), Color("4d5b60"))
			for rung in range(6):
				box(root, "LadderRung", Vector3(0.34, 0.035, 0.04), Vector3(width * 0.22, y + rung * 0.23, 2.13), Color("56646a"))
			for side in [-1, 1]:
				box(root, "LadderStringer", Vector3(0.035, 1.40, 0.04), Vector3(width * 0.22 + side * 0.19, y + 0.70, 2.13), Color("56646a"))
	box(root, "Storefront", Vector3(width * 0.72, 0.57, 0.05), Vector3(0, 0.43, 1.79), Color("2e555c"))
	box(root, "StripedAwning", Vector3(width * 0.86, 0.12, 0.70), Vector3(0, 0.87, 1.95), Color("a2655a"))

static func label(parent: Node3D, text: String, pos: Vector3, pixel_size: float, color: Color) -> void:
	var node := Label3D.new()
	node.text = text
	node.name = "Sign_" + text.replace(" ", "_")
	node.position = pos
	node.font_size = 40
	node.pixel_size = pixel_size
	node.modulate = color
	node.outline_size = 2
	parent.add_child(node)

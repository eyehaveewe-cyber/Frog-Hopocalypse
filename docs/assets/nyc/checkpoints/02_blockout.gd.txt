extends RefCounted
## Original low-poly street kit. All vehicles face +X. Wheel pivots rotate on Z.
static var meshes: Dictionary = {}
static var materials: Dictionary = {}

static func material(color: Color, emission: float = 0.0) -> StandardMaterial3D:
	var key := str(color) + str(emission)
	if not materials.has(key):
		var value := StandardMaterial3D.new()
		value.albedo_color = color
		value.roughness = 0.9
		value.metallic_specular = 0.0
		value.shading_mode = BaseMaterial3D.SHADING_MODE_PER_VERTEX
		if emission > 0:
			value.emission_enabled = true
			value.emission = color
			value.emission_energy_multiplier = emission
		materials[key] = value
	return materials[key]

static func box(parent: Node3D, title: String, size: Vector3, pos: Vector3, color: Color, glow: float = 0.0) -> MeshInstance3D:
	var key := str(size)
	if not meshes.has(key):
		var mesh := BoxMesh.new()
		mesh.size = size
		meshes[key] = mesh
	var piece := MeshInstance3D.new()
	piece.name = title
	piece.mesh = meshes[key]
	piece.material_override = material(color, glow)
	piece.position = pos
	parent.add_child(piece)
	return piece

static func cylinder(parent: Node3D, title: String, radius: float, height: float, pos: Vector3, color: Color, top: float = -1.0) -> MeshInstance3D:
	var key := "cylinder" + str([radius, height, top])
	if not meshes.has(key):
		var mesh := CylinderMesh.new()
		mesh.bottom_radius = radius
		mesh.top_radius = radius if top < 0 else top
		mesh.height = height
		mesh.radial_segments = 8
		meshes[key] = mesh
	var piece := MeshInstance3D.new()
	piece.name = title
	piece.mesh = meshes[key]
	piece.material_override = material(color)
	piece.position = pos
	parent.add_child(piece)
	return piece

static func vehicle_length(kind: String) -> float:
	return 4.4 if kind == "bus" else (3.5 if kind == "truck" else 2.6)

static func vehicle(kind: String, variant: int = 0) -> Node3D:
	var root := Node3D.new()
	root.name = "NYC_" + kind
	var length := vehicle_length(kind)
	var paint := Color("efbc32") if kind == "taxi" else Color("375f84")
	if kind == "bus":
		paint = Color("dce2d9")
	elif kind == "truck":
		paint = Color("be7860")
	elif kind == "sedan":
		paint = [Color("375f84"), Color("bc5a58"), Color("7c947d"), Color("c8c1a4")][posmod(variant, 4)]
	box(root, "Chassis", Vector3(length - 0.1, 0.22, 1.0), Vector3(0, 0.27, 0), Color("252a35"))
	box(root, "Body", Vector3(length, 0.48, 1.08), Vector3(0, 0.54, 0), paint)
	if kind == "bus":
		box(root, "PassengerBody", Vector3(length - 0.16, 0.92, 1.05), Vector3(0, 1.13, 0), paint)
	elif kind == "truck":
		box(root, "CargoBox", Vector3(2.25, 1.08, 1.08), Vector3(-0.53, 1.03, 0), Color("d4cbb4"))
		box(root, "Cab", Vector3(1.03, 0.75, 1.04), Vector3(1.04, 0.90, 0), paint)
	else:
		box(root, "Cabin", Vector3(1.32, 0.50, 0.94), Vector3(-0.1, 0.97, 0), paint.lightened(0.08))
	return root

static func scenery(kind: String) -> Node3D:
	var root := Node3D.new()
	root.name = "NYC_" + kind
	box(root, "StreetProp", Vector3(0.45, 0.8, 0.45), Vector3(0, 0.4, 0), Color("657878"))
	return root

static func district(root: Node3D, width: int, height: int, cell: float) -> void:
	var half_width := width * cell * 0.5
	var back := -height * cell * 0.5 - 3.0
	for i in range(-4, 5):
		var building_height := 4.0 + posmod(i * 7 + 17, 6) * 1.1
		box(root, "Building_%d" % i, Vector3(4.0, building_height, 3.6), Vector3(i * 4.5, building_height * 0.5, back), Color("586178"))
	for side in [-1, 1]:
		for i in range(3):
			box(root, "SideBuilding", Vector3(3.0, 4.0 + i, 3.5), Vector3(side * (half_width + 2.7), 2.0 + i * 0.5, back + 5.0 + i * 4.0), Color("8a685f"))
